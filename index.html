      if (!state.qaVideoFile) {
        showToast('กรุณานำเข้าคลิปวิดีโอก่อนครับ');
        return;
      }

      const btn = document.getElementById('runAuditBtn');
      const box = document.getElementById('qaVerdictBox');

      // ถ้ายังไม่มีเฟรมภาพ ลองจับภาพจากตัวเล่นวิดีโออีกครั้ง
      if (!state.extractedFrames.length) {
        captureCurrentPreviewFrameFallback();
      }

      // กรณีพร้อมใช้งาน Gemini Vision ครบทั้ง 3 อย่าง
      if (state.geminiKey && state.productImages.length > 0 && state.extractedFrames.length > 0) {
        btn.disabled = true;
        btn.textContent = '⏳ AI กำลังตรวจเทียบทุกมิติ...';
        box.className = 'rounded-2xl p-4 border border-pink-200 bg-pink-50 text-sm space-y-1';
        box.innerHTML = `
⏳ กำลังส่งภาพสินค้าจริง (${state.productImages.length} รูป) และเฟรมวิดีโอ (${state.extractedFrames.length} เฟรม) ให้ AI วิเคราะห์...
`;

        try {
          const audit = await auditWithGeminiVision();
          ['shape', 'label', 'logo', 'color', 'texture', 'presenter'].forEach((k, idx) => {
            if (audit.dimensions && audit.dimensions[k]) {
              state.qaDimensions[idx].status = audit.dimensions[k].pass ? 'pass' : 'fail';
              state.qaDimensions[idx].note = audit.dimensions[k].reason || '';
            }
          });
          renderQaDimensions();
          computeFinalVerdict(audit.fixSuggestion);
          showToast('AI วิเคราะห์ความตรงปกครบ 6 มิติเรียบร้อย!');
        } catch (err) {
          showToast('AI Audit ขัดข้อง: ' + err.message);
          computeFinalVerdict();
          box.innerHTML += `
⚠ เรียก AI ไม่สำเร็จ: ${escapeHtml(err.message)} (กรุณาตรวจสอบ Gemini API Key หรือชื่อโมเดล แล้วลองใหม่ หรือกดประเมิน 6 มิติด้วยตนเอง)
`;
        } finally {
          btn.disabled = false;
          btn.textContent = '🔍 เริ่มตรวจสอบความตรงปกทุกมิติ';
        }
        return;
      }

      // กรณีเงื่อนไขยังไม่ครบ ให้แจ้งชัดเจนในกล่องผลลัพธ์ (ปุ่มจะไม่เงียบ)
      computeFinalVerdict();
      const missingReasons = [];
